# ── v5.4.5: corporate-action watchdog ──────────────────────────────
    def _detect_basis_changes(self, cache: pl.DataFrame, fresh: pl.DataFrame) -> List[str]:
        """Compare overlapping (symbol,date) closes: a stable ratio != 1
        (split/dividend re-basis by yfinance) or unstable ratio (restatement
        / mixed-source rows) → symbol needs full-history refetch.
        F4: detects 2+ session variance or 1-session >1% action."""
        if cache is None or cache.is_empty() or fresh is None or fresh.is_empty():
            return []
        overlap = (fresh.select(["symbol","date","close"])
                        .join(cache.select(["symbol","date","close"]),
                              on=["symbol","date"], how="inner", suffix="_old"))
        if overlap.is_empty(): return []
        overlap = overlap.with_columns((pl.col("close")/pl.col("close_old")).alias("_r"))
        flags = (overlap.group_by("symbol").agg([
                    pl.col("_r").min().alias("_mn"), pl.col("_r").max().alias("_mx"),
                    pl.col("_r").median().alias("_med"), pl.len().alias("_n")])
                 .filter(((pl.col("_n") >= 2) &
                          ( ((pl.col("_mx")-pl.col("_mn")) < 0.01) &
                            ((pl.col("_med")-1.0).abs() > 0.005)
                            | ((pl.col("_mx")-pl.col("_mn")) >= 0.01) ))
                         | ((pl.col("_n") == 1) & ((pl.col("_med")-1.0).abs() > 0.01)))
                 ["symbol"].to_list())
        if flags:
            _dp_logger.info(f"Adjustment watchdog: basis change in {len(flags)} symbol(s) — "
                            f"refetching full history: {', '.join(flags[:12])}")
        return flags
